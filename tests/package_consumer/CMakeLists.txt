cmake_minimum_required(VERSION 3.5)
project(gpr_external_client C CXX)
find_package(GPR CONFIG REQUIRED COMPONENTS ${EXPECTED_COMPONENTS})
enable_testing()
set(definitions "")
set(libraries "")
foreach(component encoder decoder capture preview sdk)
  string(TOUPPER ${component} upper)
  list(FIND EXPECTED_COMPONENTS ${component} expected)
  if(TARGET GPR::${component})
    if(expected EQUAL -1)
      message(FATAL_ERROR "Unexpected component leaked into package: ${component}")
    endif()
    list(APPEND definitions HAS_${upper}=1)
    list(APPEND libraries GPR::${component})
  else()
    if(NOT expected EQUAL -1)
      message(FATAL_ERROR "Expected component missing: ${component}")
    endif()
    list(APPEND definitions HAS_${upper}=0)
  endif()
endforeach()
if(TARGET GPR::capture OR TARGET GPR::preview OR TARGET GPR::encoder OR TARGET GPR::decoder)
  add_executable(native_client main.cpp headers.c)
  target_compile_features(native_client PRIVATE cxx_std_11)
  target_compile_definitions(native_client PRIVATE ${definitions} NATIVE_CLIENT=1)
  # SDK closure is tested separately, without other targets masking a missing dependency.
  list(REMOVE_ITEM libraries GPR::sdk)
  target_link_libraries(native_client PRIVATE ${libraries})
  add_test(NAME installed_native_client COMMAND native_client ${FIXTURES})
endif()
if(TARGET GPR::sdk)
  add_executable(sdk_client main.cpp headers.c)
  target_compile_features(sdk_client PRIVATE cxx_std_11)
  target_compile_definitions(sdk_client PRIVATE HAS_SDK=1 HAS_ENCODER=0 HAS_DECODER=0 HAS_CAPTURE=0 HAS_PREVIEW=0 NATIVE_CLIENT=0)
  target_link_libraries(sdk_client PRIVATE GPR::sdk)
  add_test(NAME installed_sdk_client COMMAND sdk_client ${FIXTURES} ${CMAKE_CURRENT_BINARY_DIR}/sample.gpr)
endif()
